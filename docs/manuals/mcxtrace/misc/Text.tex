\section{The \texttt{Text} McXtrace Component}
Release: McXtrace 3.9

Draw a text

\subsection*{Identification}
\begin{itemize}
  \item \textbf{Author:} Emmanuel Farhi

  \item \textbf{Date:} Oct 10, 2026
\end{itemize}

\subsection*{Description}
A text with given size and orientation. The text is drawn in upper letters.

Example: Text(text="SOLEIL", size=0.5, plane="XY")

\subsection*{Input parameters}
Parameters in \textbf{boldface} are required; the others are optional.

\begin{longtable}{p{0.22\textwidth}p{0.12\textwidth}p{0.46\textwidth}p{0.14\textwidth}}
\toprule
\textbf{Name} & \textbf{Unit} & \textbf{Description} & \textbf{Default} \\
\midrule
\endhead
\textbf{text} & str & Text to draw. &  \\
size & m & Size of text. & 0.5 \\
plane & str & "XY","YZ",or "XZ" & "XY" \\
pen\_thickness & 1 & Stroke diameter as fraction of text size. 0 draws thin lines, \textgreater{}0 draws each stroke as a cylinder with spherical joints. & 0.1 \\
\bottomrule
\end{longtable}
